Question 1: NGINX TLS Configuration An NGINX Deployment named nginx-static is running in the nginx-static namespace. It is configured using a ConfigMap named nginx-config. Task Update the nginx-config ConfigMap to allow only TLSv1.3 connections. Re-create, restart, or scale resources as necessary. Verify (should FAIL as TLSv1.2 is no longer allowed): curl -k --tls-max 1.2 https://web.k8s.local/ and should still work: curl -k --tlsv1.3 https://web.k8s.local/ (-k because the lab certificate is self-signed)

Show answer
# 1. Edit the ConfigMap: keep only TLSv1.3
k -n nginx-static edit cm nginx-config
#     ssl_protocols TLSv1.3;

# 2. nginx reads the config only at start -> restart the pods
k -n nginx-static rollout restart deploy nginx-static
k -n nginx-static rollout status deploy nginx-static

# 3. Verify
curl -k --tls-max 1.2 https://web.k8s.local/   # error: tlsv1 alert protocol version
curl -k --tlsv1.3 https://web.k8s.local/       # Hello from nginx
What the setup creates
#!/bin/bash
set -e
kubectl create ns nginx-static --dry-run=client -o yaml | kubectl apply -f -
D=$(mktemp -d)
openssl req -x509 -nodes -days 90 -newkey rsa:2048 -keyout $D/tls.key -out $D/tls.crt \
  -subj "/CN=web.k8s.local" -addext "subjectAltName=DNS:web.k8s.local" 2>/dev/null
kubectl -n nginx-static create secret tls nginx-tls-secret --cert=$D/tls.crt --key=$D/tls.key \
  --dry-run=client -o yaml | kubectl apply -f -
rm -rf $D
kubectl apply -f - <<'EOF'
apiVersion: v1
kind: ConfigMap
metadata:
  name: nginx-config
  namespace: nginx-static
data:
  default.conf: |
    server {
      listen 443 ssl;
      server_name localhost;

      ssl_protocols TLSv1.2 TLSv1.3;
      ssl_certificate     /etc/nginx/ssl/tls.crt;
      ssl_certificate_key /etc/nginx/ssl/tls.key;

      location / {
        return 200 'Hello from nginx';
      }
    }
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nginx-static
  namespace: nginx-static
spec:
  replicas: 1
  selector:
    matchLabels:
      app: nginx-static
  template:
    metadata:
      labels:
        app: nginx-static
    spec:
      containers:
      - name: nginx
        image: nginx:1.24
        volumeMounts:
        - name: config
          mountPath: /etc/nginx/conf.d
        - name: tls
          mountPath: /etc/nginx/ssl
      volumes:
      - name: config
        configMap:
          name: nginx-config
      - name: tls
        secret:
          secretName: nginx-tls-secret
---
apiVersion: v1
kind: Service
metadata:
  name: nginx-service
  namespace: nginx-static
spec:
  type: NodePort
  selector:
    app: nginx-static
  ports:
  - port: 443
    targetPort: 443
    nodePort: 31337
EOF
kubectl -n nginx-static rollout status deploy nginx-static --timeout=180s
IP=$(kubectl -n nginx-static get svc nginx-service -o jsonpath='{.spec.clusterIP}')
sudo sed -i '/web.k8s.local/d' /etc/hosts
echo "$IP web.k8s.local" | sudo tee -a /etc/hosts >/dev/null
echo "[OK] /etc/hosts: $IP web.k8s.local   (try: curl -k https://web.k8s.local/)"

cat <<'EOF' > gw.yaml
apiVersion: gateway.networking.k8s.io/v1
kind: Gateway
metadata:
  name: web-gateway
spec:
  gatewayClassName: nginx
  listeners:
  - name: https
    protocol: HTTPS
    port: 443
    hostname: gateway.web.k8s.local
    tls:
      mode: Terminate
      certificateRefs:
      - kind: Secret
        name: gw-api-tls-secret
---
apiVersion: gateway.networking.k8s.io/v1
kind: HTTPRoute
metadata:
  name: web-route
spec:
  parentRefs:
  - name: web-gateway
  hostnames:
  - gateway.web.k8s.local
  rules:
  - matches:
    - path:
        type: PathPrefix
        value: /
    backendRefs:
    - name: webapp
      port: 80
EOF
k apply -f gw.yaml
k delete ingress web
k get gateway,httproute
# (repo uses port 8443 only because its Traefik controller listens there; the exam wants 443)
What the setup creates
#!/bin/bash
set -e
echo "Installing Gateway API CRDs (v1.3.0)..."
kubectl apply -f https://github.com/kubernetes-sigs/gateway-api/releases/download/v1.3.0/standard-install.yaml >/dev/null
D=$(mktemp -d)
openssl req -x509 -nodes -days 90 -newkey rsa:2048 -keyout $D/tls.key -out $D/tls.crt \
  -subj "/CN=gateway.web.k8s.local" -addext "subjectAltName=DNS:gateway.web.k8s.local" 2>/dev/null
kubectl create secret tls gw-api-tls-secret --cert=$D/tls.crt --key=$D/tls.key --dry-run=client -o yaml | kubectl apply -f -
rm -rf $D
kubectl apply -f - <<'EOF'
apiVersion: apps/v1
kind: Deployment
metadata:
  name: webapp
spec:
  replicas: 1
  selector:
    matchLabels:
      app: webapp
  template:
    metadata:
      labels:
        app: webapp
    spec:
      containers:
      - name: web
        image: nginx
        ports:
        - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: webapp
spec:
  selector:
    app: webapp
  ports:
  - port: 80
    targetPort: 80
---
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: web
  annotations:
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
spec:
  ingressClassName: nginx
  tls:
  - hosts:
    - gateway.web.k8s.local
    secretName: gw-api-tls-secret
  rules:
  - host: gateway.web.k8s.local
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: webapp
            port:
              number: 80
---
apiVersion: gateway.networking.k8s.io/v1
kind: GatewayClass
metadata:
  name: nginx
spec:
  controllerName: example.net/nginx-gateway-controller
EOF
echo "[OK] Ingress 'web', Service 'webapp', Secret 'gw-api-tls-secret', GatewayClass 'nginx' created"

# 'kubectl autoscale' cannot set the stabilization window -> start from it, then add behavior
k -n autoscale autoscale deploy apache-server --name=apache-server --cpu-percent=50 --min=1 --max=4 \
  --dry-run=client -o yaml > hpa.yaml
# or write it directly:
cat <<'EOF' > hpa.yaml
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: apache-server
  namespace: autoscale
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: apache-server
  minReplicas: 1
  maxReplicas: 4
  metrics:
  - type: Resource
    resource:
      name: cpu
      target:
        type: Utilization
        averageUtilization: 50
  behavior:
    scaleDown:
      stabilizationWindowSeconds: 30
EOF
k apply -f hpa.yaml
k -n autoscale get hpa          # TARGETS cpu: 0%/50% after ~1 min

# template (no CRDs) -> /argo-helm.yaml   (note: /argo-helm.yaml, not /root/)

  --set crds.install=false

# --skip-crds does NOT help here: this chart ships CRDs as templates controlled by crds.install
What the setup creates
#!/bin/bash
set -e
if ! command -v helm >/dev/null; then
  echo "Installing helm..."; curl -fsSL https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash
fi
echo "Pre-installing the Argo CD CRDs (v2.13.0)..."
for u in https://raw.githubusercontent.com/argoproj/argo-cd/v2.13.0/manifests/crds/application-crd.yaml https://raw.githubusercontent.com/argoproj/argo-cd/v2.13.0/manifests/crds/applicationset-crd.yaml https://raw.githubusercontent.com/argoproj/argo-cd/v2.13.0/manifests/crds/appproject-crd.yaml; do kubectl apply --server-side --force-conflicts -f "$u" >/dev/null; done
kubectl get crd | grep argoproj.io
helm repo remove argo >/dev/null 2>&1 || true
rm -f /argo-helm.yaml
echo "[OK] Argo CD CRDs are pre-installed. helm repo 'argo' is not added yet."
